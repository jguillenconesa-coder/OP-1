cmake_minimum_required(VERSION 3.22)
project(PocketOne VERSION 0.4.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
include(CTest)
add_executable(engine_tests Tests/EngineTests.cpp)
target_include_directories(engine_tests PRIVATE Source)
add_test(NAME engine COMMAND engine_tests)
add_executable(tape_tests Tests/TapeTests.cpp)
target_include_directories(tape_tests PRIVATE Source)
add_test(NAME tape COMMAND tape_tests)
add_executable(sequencer_tests Tests/SequencerTests.cpp)
target_include_directories(sequencer_tests PRIVATE Source)
add_test(NAME sequencer COMMAND sequencer_tests)
find_package(Python3 3.11 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND AND BUILD_TESTING)
  add_test(NAME firmware_inventory COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/Tests/FirmwareInventoryTests.py")
endif()
option(POCKET_BUILD_PLUGIN "Build VST3 and standalone instrument" ON)
if(POCKET_BUILD_PLUGIN)
  if(DEFINED JUCE_PACKAGE_PATH)
    find_package(JUCE CONFIG REQUIRED PATHS "${JUCE_PACKAGE_PATH}" NO_DEFAULT_PATH)
  elseif(DEFINED JUCE_PATH)
    add_subdirectory("${JUCE_PATH}" "${CMAKE_BINARY_DIR}/JUCE")
  else()
    include(FetchContent)
    FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 51a8a6d7aeae7326956d747737ccf1575e61e209 GIT_SHALLOW FALSE)
    FetchContent_MakeAvailable(JUCE)
  endif()
  if(MINGW)
    message(FATAL_ERROR "JUCE 8 does not support MinGW. Build with MSVC on Windows (see docs/windows.md).")
  endif()
  set(_pocket_manifest TRUE)
  if(CMAKE_CROSSCOMPILING)
    set(_pocket_manifest FALSE)
  endif()
  juce_add_plugin(PocketOne COMPANY_NAME "Pocket Instruments" BUNDLE_ID "studio.pocket.one"
    PLUGIN_MANUFACTURER_CODE Pckt PLUGIN_CODE Pko1
    FORMATS VST3 Standalone PRODUCT_NAME "Pocket One"
    IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE
    COPY_PLUGIN_AFTER_BUILD FALSE VST3_AUTO_MANIFEST ${_pocket_manifest})
  file(GLOB POCKET_FACTORY_WAVS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/Assets/Factory/*.wav")
  juce_add_binary_data(PocketFactoryData HEADER_NAME FactoryData.h NAMESPACE FactoryData SOURCES ${POCKET_FACTORY_WAVS})
  set_target_properties(PocketFactoryData PROPERTIES POSITION_INDEPENDENT_CODE TRUE)
  target_link_libraries(PocketOne PRIVATE PocketFactoryData)
  target_sources(PocketOne PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp Source/UserBank.cpp Source/ScreenAnimations.cpp Source/SoundLibrary.cpp Source/SoundBrowser.cpp Source/TapeControls.cpp Source/InputCapture.cpp)
  if(WIN32 AND CMAKE_CROSSCOMPILING)
    # llvm-mt cannot inject manifests into PE files. Embed the dependency as an
    # explicit RC resource instead; /manifest:no must never drop Common Controls.
    set(_pocket_resources "${CMAKE_CURRENT_SOURCE_DIR}/packaging/windows/resources")
    target_sources(PocketOne_Standalone PRIVATE "${_pocket_resources}/standalone.rc")
    target_sources(PocketOne_VST3 PRIVATE "${_pocket_resources}/plugin.rc")
    target_include_directories(PocketOne_Standalone PRIVATE "${_pocket_resources}")
    target_include_directories(PocketOne_VST3 PRIVATE "${_pocket_resources}")
  endif()
  target_compile_definitions(PocketOne PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
  target_link_libraries(PocketOne PRIVATE juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
endif()
if(POCKET_BUILD_PLUGIN AND BUILD_TESTING)
  add_executable(plugin_smoke Tests/PluginSmoke.cpp)
  set_target_properties(plugin_smoke PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugin_smoke_artefacts/$<CONFIG>")
  target_include_directories(plugin_smoke PRIVATE Source "$<TARGET_PROPERTY:juce::juce_audio_utils,INTERFACE_INCLUDE_DIRECTORIES>")
  target_compile_definitions(plugin_smoke PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
  target_link_libraries(plugin_smoke PRIVATE PocketOne)
  if(WIN32 AND CMAKE_CROSSCOMPILING)
    target_sources(plugin_smoke PRIVATE "${_pocket_resources}/standalone.rc")
    target_include_directories(plugin_smoke PRIVATE "${_pocket_resources}")
  endif()
  add_test(NAME plugin_smoke COMMAND plugin_smoke)
  add_executable(realtime_tests Tests/RealtimeTests.cpp)
  target_include_directories(realtime_tests PRIVATE Source "$<TARGET_PROPERTY:juce::juce_audio_utils,INTERFACE_INCLUDE_DIRECTORIES>")
  target_compile_definitions(realtime_tests PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
  target_link_libraries(realtime_tests PRIVATE PocketOne)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_compile_definitions(realtime_tests PRIVATE POCKET_TRACK_C_HEAP=1)
    target_link_options(realtime_tests PRIVATE -Wl,--wrap=malloc,--wrap=calloc,--wrap=realloc,--wrap=free)
  endif()
  if(WIN32 AND CMAKE_CROSSCOMPILING)
    target_sources(realtime_tests PRIVATE "${_pocket_resources}/standalone.rc")
    target_include_directories(realtime_tests PRIVATE "${_pocket_resources}")
  endif()
  add_test(NAME realtime COMMAND realtime_tests "${CMAKE_BINARY_DIR}/realtime-metrics.json")

  if(WIN32)
    add_executable(windows_module_probe Tests/WindowsModuleProbe.cpp)
    target_include_directories(windows_module_probe PRIVATE "${JUCE_MODULES_DIR}/juce_audio_processors/format_types/VST3_SDK")
    add_test(NAME windows_vst3_module COMMAND windows_module_probe "$<TARGET_FILE:PocketOne_VST3>")
    add_test(NAME windows_standalone_manifest COMMAND windows_module_probe --manifest-exe "$<TARGET_FILE:PocketOne_Standalone>")
  endif()
endif()
